cmake_minimum_required(VERSION 3.16)
project(WebCADShoulderDiagnostic LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCASCADE 7.8.1 EXACT CONFIG REQUIRED)
if(NOT EXISTS "${JSON_INCLUDE_DIR}/nlohmann/json.hpp")
  message(FATAL_ERROR "Pinned nlohmann-json headers required")
endif()
add_executable(ShoulderDiagnostic diagnostic.cpp)
target_include_directories(ShoulderDiagnostic PRIVATE ${OpenCASCADE_INCLUDE_DIR} ${JSON_INCLUDE_DIR})
target_link_directories(ShoulderDiagnostic PRIVATE ${OpenCASCADE_LIBRARY_DIR})
target_compile_definitions(ShoulderDiagnostic PRIVATE OCCT_STATIC_BUILD)
target_link_libraries(ShoulderDiagnostic PRIVATE "-Wl,--start-group" TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep TKGeomAlgo TKTopAlgo TKPrim TKBO TKBool TKShHealing "-Wl,--end-group" psapi winmm ws2_32)

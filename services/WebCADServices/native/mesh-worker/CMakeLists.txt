cmake_minimum_required(VERSION 3.18)
project(WebCADMeshWorker LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(opt MANIFOLD_TEST MANIFOLD_PAR MANIFOLD_PYBIND MANIFOLD_CBIND MANIFOLD_JSBIND MANIFOLD_CROSS_SECTION MANIFOLD_DOWNLOADS)
  set(${opt} OFF CACHE BOOL "" FORCE)
endforeach()
if(NOT EXISTS "${MANIFOLD_ROOT}/CMakeLists.txt" OR NOT EXISTS "${JSON_INCLUDE_DIR}/nlohmann/json.hpp")
  message(FATAL_ERROR "Provide locked Manifold 3.5.4 and nlohmann-json 3.11.3 dependency directories")
endif()
add_subdirectory("${MANIFOLD_ROOT}" "${CMAKE_BINARY_DIR}/manifold" EXCLUDE_FROM_ALL)
add_executable(WebCADMeshWorker worker.cpp)
target_include_directories(WebCADMeshWorker PRIVATE "${JSON_INCLUDE_DIR}")
target_link_libraries(WebCADMeshWorker PRIVATE manifold bcrypt psapi)
if(MINGW)
  target_link_options(WebCADMeshWorker PRIVATE -static)
endif()
install(TARGETS WebCADMeshWorker RUNTIME DESTINATION bin)

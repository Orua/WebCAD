cmake_minimum_required(VERSION 3.20)
project(WebCADBenchmarkStepReader LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCASCADE 7.8.1 EXACT CONFIG REQUIRED)
set(ORACLE_OCCT_INCLUDE_DIR "${OpenCASCADE_INCLUDE_DIR}" CACHE PATH "Installed OCCT headers including STEP data exchange")
set(ORACLE_OCCT_LIBRARY_DIR "${OpenCASCADE_LIBRARY_DIR}" CACHE PATH "Installed OCCT libraries including TKDESTEP")
set(JSON_INCLUDE_DIR "" CACHE PATH "Directory containing nlohmann/json.hpp")
if(NOT EXISTS "${JSON_INCLUDE_DIR}/nlohmann/json.hpp")
 message(FATAL_ERROR "Set JSON_INCLUDE_DIR to the existing nlohmann-json headers")
endif()
if(NOT EXISTS "${ORACLE_OCCT_INCLUDE_DIR}/STEPControl_Reader.hxx")
 message(FATAL_ERROR "OCCT SDK lacks STEPControl_Reader.hxx: install DataExchange headers/libraries into an isolated oracle SDK")
endif()
find_library(ORACLE_STEP_LIBRARY NAMES TKDESTEP PATHS "${ORACLE_OCCT_LIBRARY_DIR}" NO_DEFAULT_PATH REQUIRED)
add_executable(WebCADBenchmarkStepReader step-reader.cpp)
target_include_directories(WebCADBenchmarkStepReader PRIVATE "${ORACLE_OCCT_INCLUDE_DIR}" "${JSON_INCLUDE_DIR}")
target_link_directories(WebCADBenchmarkStepReader PRIVATE "${ORACLE_OCCT_LIBRARY_DIR}")
option(ORACLE_OCCT_STATIC "Link an existing static OCCT SDK" ON)
if(ORACLE_OCCT_STATIC)
 target_compile_definitions(WebCADBenchmarkStepReader PRIVATE OCCT_STATIC_BUILD)
endif()
set(ORACLE_LIBRARIES TKDESTEP TKDE TKXSBase TKXCAF TKVCAF TKCAF TKLCAF TKCDF TKV3d TKService TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep TKGeomAlgo TKTopAlgo TKPrim TKBO TKBool TKShHealing TKMesh TKHLR)
if(MINGW AND ORACLE_OCCT_STATIC)
 target_link_libraries(WebCADBenchmarkStepReader PRIVATE "-Wl,--start-group" ${ORACLE_LIBRARIES} "-Wl,--end-group" winmm ws2_32 gdi32 user32 advapi32 ole32 shell32 comdlg32)
else()
 target_link_libraries(WebCADBenchmarkStepReader PRIVATE ${ORACLE_LIBRARIES})
endif()
# No installation, Services linkage, runtime server, network access or source-tree output.
